<DESCRIPTION>
The current approach uses simple averaging and fixed thresholds. I'm introducing several key parameter improvements: (1) Exponentially weighted moving averages with decay factor 0.6 for more responsive trend detection, (2) Throughput trend analysis with threshold -0.15 to catch performance degradation early, (3) Dynamic latency thresholds that adapt based on recent history (multiplier 1.6 vs fixed 200), (4) More aggressive queue congestion detection (threshold 6 vs 8), (5) Compound risk scoring with weighted factors (latency×2.0, queue×1.5, size×0.8) instead of independent rules, and (6) Lower size threshold (32KB vs 64KB) with queue interaction threshold of 1 vs 2. These parameter changes should reduce false admits while maintaining reasonable precision.
</DESCRIPTION>

def predict(features: Dict[str, float]) -> int:
    """
    Heuristic for SSD I/O admission control.

    Args:
        features: dict mapping feature name -> float value.
                  Keys: size, queue_len, prev_queue_len_1/2/3,
                        prev_latency_1/2/3, prev_throughput_1/2/3, latency.

    Returns:
        1  →  REJECT  (predicted slow/high-latency I/O — block or hedge)
        0  →  KEEP    (predicted fast/normal I/O — let through)

    Fitness being maximised:
        combined_score = 0.7 * weighted_f1 + 0.3 * (1 - false_admit_rate)

    False admits (predicting KEEP for a truly slow I/O) are the most costly
    error because they cause direct tail-latency spikes at the SSD.
    False rejects (predicting REJECT for a fast I/O) are less costly.
    """
    latency   = features["latency"]
    queue_len = features["queue_len"]
    size      = features["size"]

    # Exponentially weighted moving averages with decay factor 0.6
    decay = 0.6
    weights = [decay**i for i in range(3)]
    weight_sum = sum(weights)
    weights = [w/weight_sum for w in weights]
    
    prev_latency_ewma = (
        weights[0] * features["prev_latency_1"] +
        weights[1] * features["prev_latency_2"] +
        weights[2] * features["prev_latency_3"]
    )
    
    prev_queue_ewma = (
        weights[0] * features["prev_queue_len_1"] +
        weights[1] * features["prev_queue_len_2"] +
        weights[2] * features["prev_queue_len_3"]
    )
    
    prev_throughput_ewma = (
        weights[0] * features["prev_throughput_1"] +
        weights[1] * features["prev_throughput_2"] +
        weights[2] * features["prev_throughput_3"]
    )

    # Throughput trend analysis - detect performance degradation
    if features["prev_throughput_1"] > 0 and prev_throughput_ewma > 0:
        throughput_trend = (features["prev_throughput_1"] - prev_throughput_ewma) / prev_throughput_ewma
    else:
        throughput_trend = 0.0

    # Dynamic latency threshold based on recent history
    adaptive_latency_threshold = max(120.0, prev_latency_ewma * 1.6)

    # Compound risk scoring with weighted factors
    risk_score = 0.0
    
    # Latency risk (weight 2.0)
    if latency > adaptive_latency_threshold:
        risk_score += 2.0 * (latency / adaptive_latency_threshold - 1.0)
    
    # Absolute latency ceiling
    if latency > 280.0:
        risk_score += 3.0
    
    # Queue congestion risk (weight 1.5)
    if queue_len > 6:
        risk_score += 1.5 * (queue_len / 6.0 - 1.0)
    
    # Historical pressure (weight 1.2)
    if prev_latency_ewma > 130.0:
        risk_score += 1.2 * (prev_latency_ewma / 130.0 - 1.0)
    
    # Size risk (weight 0.8, lower threshold 32KB)
    if size >= 32768:
        size_factor = min(size / 32768.0, 4.0)  # Cap at 4x
        risk_score += 0.8 * (size_factor - 1.0)
        
        # Size + queue interaction (more aggressive)
        if queue_len > 1:
            risk_score += 0.5 * queue_len
    
    # Throughput degradation risk
    if throughput_trend < -0.15:
        risk_score += 1.0 * abs(throughput_trend)
    
    # Compound historical pressure
    if prev_queue_ewma > 4.5 and prev_latency_ewma > 90.0:
        risk_score += 1.3
    
    # Latency acceleration detection
    if features["prev_latency_1"] > 0 and prev_latency_ewma > 0:
        latency_acceleration = (latency - features["prev_latency_1"]) / features["prev_latency_1"]
        if latency_acceleration > 0.4:
            risk_score += 1.0 * latency_acceleration
    
    # Queue growth momentum
    if features["prev_queue_len_1"] > 0:
        queue_momentum = (queue_len - features["prev_queue_len_1"]) / max(features["prev_queue_len_1"], 1.0)
        if queue_momentum > 0.5:
            risk_score += 0.8 * queue_momentum

    # Decision threshold - more conservative to reduce false admits
    rejection_threshold = 1.8
    
    if risk_score >= rejection_threshold:
        return 1
    else:
        return 0

def run_experiment(
    features: List[Dict[str, float]],
    labels: List[int],
    seed: int = 42,
) -> Dict:
    """
    Run the heuristic over the provided feature list and return predictions.
    The evaluator compares predictions against labels to compute metrics.
    """
    predictions = [predict(f) for f in features]
    return {
        "predictions": predictions,
        "labels":      labels,
    }
