def predict(features: Dict[str, float]) -> int:
    """
    Heuristic for SSD I/O admission control.

    Args:
        features: dict mapping feature name -> float value.
                  Keys: size, queue_len, prev_queue_len_1/2/3,
                        prev_latency_1/2/3, prev_throughput_1/2/3, latency.

    Returns:
        1  →  REJECT  (predicted slow/high-latency I/O — block or hedge)
        0  →  KEEP    (predicted fast/normal I/O — let through)

    Fitness being maximised:
        combined_score = 0.7 * weighted_f1 + 0.3 * (1 - false_admit_rate)

    False admits (predicting KEEP for a truly slow I/O) are the most costly
    error because they cause direct tail-latency spikes at the SSD.
    False rejects (predicting REJECT for a fast I/O) are less costly.
    """
    latency = features["latency"]
    queue_len = features["queue_len"]
    size = features["size"]
    
    # Extract historical values
    prev_latencies = [
        features["prev_latency_1"],
        features["prev_latency_2"],
        features["prev_latency_3"]
    ]
    
    prev_queues = [
        features["prev_queue_len_1"],
        features["prev_queue_len_2"],
        features["prev_queue_len_3"]
    ]
    
    prev_throughputs = [
        features["prev_throughput_1"],
        features["prev_throughput_2"],
        features["prev_throughput_3"]
    ]
    
    # Compute exponentially weighted moving averages (more weight on recent)
    weights = [0.5, 0.3, 0.2]  # Sum = 1.0
    
    ewma_latency = sum(w * lat for w, lat in zip(weights, prev_latencies))
    ewma_queue = sum(w * q for w, q in zip(weights, prev_queues))
    ewma_throughput = sum(w * tp for w, tp in zip(weights, prev_throughputs))
    
    # Compute latency trend (acceleration/deceleration)
    if prev_latencies[1] > 0:
        latency_slope = (prev_latencies[0] - prev_latencies[1]) / prev_latencies[1]
    else:
        latency_slope = 0
    
    # Compute queue trend
    if prev_queues[1] > 0:
        queue_slope = (prev_queues[0] - prev_queues[1]) / max(prev_queues[1], 1)
    else:
        queue_slope = 0
    
    # Compute throughput trend (negative slope = degrading performance)
    if prev_throughputs[1] > 0:
        throughput_slope = (prev_throughputs[0] - prev_throughputs[1]) / prev_throughputs[1]
    else:
        throughput_slope = 0
    
    # Multi-dimensional pressure scoring
    pressure_score = 0.0
    
    # 1. Current latency pressure (most immediate signal)
    if latency > 180:
        pressure_score += 4.0
    elif latency > 120:
        pressure_score += 2.0
    elif latency > 80:
        pressure_score += 1.0
    
    # 2. Queue congestion pressure
    if queue_len > 10:
        pressure_score += 3.5
    elif queue_len > 6:
        pressure_score += 2.0
    elif queue_len > 3:
        pressure_score += 1.0
    
    # 3. Historical latency pressure (with trend weighting)
    if ewma_latency > 140:
        pressure_score += 2.5
    elif ewma_latency > 100:
        pressure_score += 1.5
    elif ewma_latency > 70:
        pressure_score += 0.8
    
    # 4. Trend-based pressure adjustments
    if latency_slope > 0.3:  # Latency accelerating
        pressure_score += 1.5
    elif latency_slope > 0.1:
        pressure_score += 0.7
    
    if queue_slope > 0.5:  # Queue growing rapidly
        pressure_score += 1.2
    elif queue_slope > 0.2:
        pressure_score += 0.6
    
    if throughput_slope < -0.2:  # Throughput degrading
        pressure_score += 1.0
    elif throughput_slope < -0.1:
        pressure_score += 0.5
    
    # 5. Size-based risk multiplier
    size_multiplier = 1.0
    if size >= 131072:  # 128KB+
        size_multiplier = 1.4
    elif size >= 65536:  # 64KB+
        size_multiplier = 1.2
    elif size >= 32768:  # 32KB+
        size_multiplier = 1.1
    
    # Apply size multiplier
    pressure_score *= size_multiplier
    
    # 6. Compound risk factors
    if ewma_queue > 5 and ewma_latency > 90:
        pressure_score += 1.5  # Sustained congestion
    
    if queue_len > 4 and latency > 100:
        pressure_score += 1.2  # Current stress
    
    if latency_slope > 0.2 and queue_slope > 0.3:
        pressure_score += 1.0  # Deteriorating trends
    
    # Adaptive threshold based on system state
    base_threshold = 2.8
    
    # Lower threshold when system is already stressed
    if ewma_latency > 120 or ewma_queue > 7:
        threshold = base_threshold - 0.8
    elif ewma_latency > 80 or ewma_queue > 4:
        threshold = base_threshold - 0.4
    else:
        threshold = base_threshold
    
    # Final decision
    if pressure_score >= threshold:
        return 1
    else:
        return 0
